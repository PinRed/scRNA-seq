	\begin{frame}{Pathway Analysis Methods}
		\textbf{Goal :} 
		\begin{enumerate}
			\item Looking at DEGs alone often can’t explain the biological mechanism. Pathway analysis, using databases like KEGG or Reactome, helps identify which pathways and functions these genes are involved in.
			
			\item It detects which pathways (e.g., cell cycle, apoptosis, immune signaling) are significantly changed between different conditions, treatments, or diseases.
			
			\item It helps summarize which cellular behaviors or biological processes may be affected based on the DEGs.
		\end{enumerate}
		
	\end{frame}

	\begin{frame}{Method Limitations}
		ORA Limitations :
		\begin{enumerate}
			\item Ignore any values associated with them such as probe intensities.
			\item Using an arbitrary threshold lead information loss.
			\item Assuming independence between genes amounts to \textbf{competitive null hypothesis testing} (see below), which ignores the correlation structure between genes.
			\item Assumes that each pathway is independent of other pathways, which is erroneous. 
		\end{enumerate}
	\end{frame}

	\begin{frame}{Pathway Topology}
		\begin{enumerate}
			\item ScorePAGE \cite{rahnenfuhrer2004calculating} : \\
			Computes the similarity between each pair of genes in a pathway. This similarity is analogous to the gene-level statistics used in FCS methods.
			\item Impact Factor (IF) analysis 
			\cite{draghici2007systems} : \\
			IF analysis models a signaling pathway
			as a graph, defines a gene-level statistic,
			called perturbation factor (PF) of a gene.
			\begin{center}
				$PF(g_i) = \Delta F(g_i) + \sum^n_{j=1} \beta_{ji} \cdot \dfrac{PF(g_j)}{N_{ds}(g_j)} $
			\end{center}
			Then Calculate IF :
			\begin{center}
				$IF(P_i) = log\left(\dfrac{1}{p_i}\right) + \dfrac{|\sum_{g \in P_i}PF(g)|}{N_{de}(P_i)}$
			\end{center}
		\end{enumerate}
	\end{frame}	
	
	\begin{frame}
		Pathways are not static, but change their interaction structure according to physiological conditions. 
		
		\vskip 0.2cm
		\textbf{NetGSA} \cite{shojaie2009analysis} accounts for the the change in correlation as well as the change in network structure as experimental conditions change.
		
		\begin{enumerate}
			\item accounts for a gene’s baseline expression.
			\item the pathways be represented as directed acyclic graphs.
		\end{enumerate}
		
	\end{frame}
